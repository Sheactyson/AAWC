# Stage 2 physical conditions. These values NEVER write residents or a target.
# CK3 min = lower bound, max = upper bound. Scores remain on a 0-100 scale.
aawc_stage2_development = { value = county.development_level min = 0 max = 100 }

aawc_stage2_main_level = {
	value = 1
	if = { limit = { OR = { has_building = castle_05 has_building = city_05 } } value = 5 }
	else_if = { limit = { OR = { has_building = castle_04 has_building = city_04 has_building = temple_04 has_building = monastery_04 has_building = temple_citadel_04 } } value = 4 }
	else_if = { limit = { OR = { has_building = castle_03 has_building = city_03 has_building = temple_03 has_building = monastery_03 has_building = temple_citadel_03 has_building = settlement_03 has_building = pirate_den_03 } } value = 3 }
	else_if = { limit = { OR = { has_building = castle_02 has_building = city_02 has_building = temple_02 has_building = monastery_02 has_building = temple_citadel_02 has_building = settlement_02 has_building = tribe_02 has_building = pirate_den_02 } } value = 2 }
}

aawc_stage2_geographic_space = {
	value = 1
	if = { limit = { agot_is_urban_terrain = yes } value = 1.15 }
	else_if = { limit = { OR = { agot_is_farmlands_terrain = yes agot_is_floodplains_terrain = yes } } value = 1.10 }
	else_if = { limit = { agot_is_plains_terrain = yes } value = 1.05 }
	else_if = { limit = { OR = { agot_is_mountains_terrain = yes agot_is_desert_mountains_terrain = yes agot_is_canyon_terrain = yes } } value = 0.65 }
	else_if = { limit = { OR = { agot_is_glacier_terrain = yes agot_is_frozen_flats_terrain = yes } } value = 0.55 }
	else_if = { limit = { agot_is_desert_terrain = yes } value = 0.70 }
	else_if = { limit = { OR = { agot_is_wetlands_terrain = yes agot_is_taiga_bog_terrain = yes agot_is_the_bog_terrain = yes } } value = 0.80 }
	else_if = { limit = { OR = { agot_is_hills_terrain = yes agot_is_highlands_terrain = yes agot_is_forest_terrain = yes agot_is_taiga_terrain = yes agot_is_cloudforest_terrain = yes } } value = 0.90 }
}

# Only housing and food scores scale physical accommodation. At 50/50 factor=1.
# One point of Habitation has four times the capacity weight of Sustenance.
aawc_stage2_accommodation_factor = {
	value = 0.25
	add = { value = aawc_stage2_structural_habitation_value multiply = 0.012 }
	add = { value = aawc_stage2_structural_sustenance_value multiply = 0.003 }
}
aawc_stage2_infrastructure_space = {
	value = 1
	if = { limit = { has_variable = aawc_stage2_space_percent } add = { value = var:aawc_stage2_space_percent divide = 100 } }
	min = 0.25
}
aawc_stage2_physical_capacity = {
	value = 0
	if = {
		limit = { aawc_population_valid_settlement_for_cohort_trigger = yes }
		value = aawc_population_holding_main_line_expected_capacity
		multiply = aawc_stage2_accommodation_factor
		multiply = aawc_stage2_geographic_space
		multiply = aawc_stage2_infrastructure_space
		min = 1
		round = yes
	}
}
aawc_stage2_capacity_adjustment = {
	value = aawc_stage2_physical_capacity
	subtract = aawc_population_holding_main_line_expected_capacity
}
aawc_stage2_occupancy = {
	value = 0
	if = {
		limit = { has_variable = aawc_population_expected var:aawc_population_expected > 0 }
		value = aawc_reforging_cohort_band_sum
		divide = var:aawc_population_expected
	}
}

	min = 0

	min = 0
