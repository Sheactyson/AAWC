aawc_lineage_post_start_baseline_effect = {
	aawc_lineage_initialize_counters_effect = yes
	if = {
		limit = { NOT = { has_global_variable = aawc_lineage_post_start_baseline_complete } }
		set_global_variable = { name = aawc_lineage_startup_pool_characters_removed value = 0 }
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE one-time post-start pool reduction begin" }
		every_living_character = {
			limit = {
				is_pool_character = yes
				is_landed = no
				is_lowborn = yes
				agot_is_dummy_character = no
				NOT = { has_character_flag = aawc_1101_generated_start_courtier }
				NOT = { any_player = { this = prev } }
			}
			change_global_variable = { name = aawc_lineage_startup_pool_characters_removed add = 1 }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE one-time cleanup removed setup pool character" }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
			death = { death_reason = death_disappearance }
		}
		# Setup pool characters can lose formal pool status before this delayed
		# pass. Remove only unattached, parentless lowborn setup wanderers. The
		# baseline completion flag guarantees that this sweep never repeats.
		every_living_character = {
			limit = {
				is_landed = no
				is_lowborn = yes
				agot_is_dummy_character = no
				NOT = { exists = employer }
				NOT = { exists = father }
				NOT = { exists = mother }
				NOT = { has_character_flag = aawc_1101_generated_start_courtier }
				NOT = { has_character_flag = aawc_lineage_known_aawc_generation }
				NOT = { has_character_flag = aawc_lineage_known_birth }
				NOT = { has_character_flag = aawc_engine_pool_reserve }
				NOT = { any_player = { this = prev } }
			}
			change_global_variable = { name = aawc_lineage_startup_wanderers_removed add = 1 }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE one-time cleanup removed unattached parentless setup wanderer" }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
			death = { death_reason = death_disappearance }
		}
		every_ruler = {
			limit = { primary_title ?= { is_mercenary_company = yes } }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE post-shatter mercenary revalidation scheduled" }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
			trigger_event = { id = aawc_startup.0003 days = 1 }
		}
		every_living_character = { add_character_flag = aawc_lineage_audited }
		set_global_variable = { name = aawc_lineage_post_start_baseline_complete value = 1 }
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE one-time post-start pool reduction completed; summary follows" }
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
	}
}
