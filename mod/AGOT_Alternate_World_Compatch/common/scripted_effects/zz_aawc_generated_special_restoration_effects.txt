aawc_apply_generated_special_restoration_from_province_effect = {
	$PROVINCE$ = {
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC RESTORE GENERATED from_province entered" }
		if = { limit = { has_holding_type = settlement_holding } error_log = "AAWC RESTORE GENERATED from_province sees settlement_holding" }
		if = { limit = { has_holding_type = ruin_holding } error_log = "AAWC RESTORE GENERATED from_province sees ruin_holding" }
		if = { limit = { has_holding_type = wilderness_holding } error_log = "AAWC RESTORE GENERATED from_province sees wilderness_holding" }
		if = { limit = { has_holding_type = unknown_holding } error_log = "AAWC RESTORE GENERATED from_province sees unknown_holding" }
		county = { aawc_apply_generated_special_restoration_effect = yes }
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC RESTORE GENERATED from_province returned from county restoration dispatch" }
	}
}
