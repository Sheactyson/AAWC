create_zanj_vassal_effect = {
	# Keep the vanilla FP3 argument signature accepted by callers without
	# creating vanilla Zanj characters in AGOT's missing faith/culture setup.
	if = {
		limit = { always = no }
		create_character = {
			gender = male
			location = $LOCATION$
			save_scope_as = new_zanj_vassal
		}
	}
}

create_zanj_courtier_effect = {
	# Keep the vanilla FP3 argument signature accepted by callers without
	# creating vanilla Zanj courtiers in AGOT's missing faith/culture setup.
	if = {
		limit = { always = no }
		create_character = {
			gender = male
			location = $LOCATION$
			save_scope_as = new_zanj_courtier
		}
		scope:new_zanj_courtier = {
			set_employer = $EMPLOYER$
		}
	}
}
