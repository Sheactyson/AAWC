aawc_cleanup_nonlandmark_default_ruins_to_wilderness_effect = {
	if = {
		limit = { aawc_tribal_shatter_active = yes }
		every_county = {
			limit = {
				any_county_province = { has_holding_type = ruin_holding }
				NOT = { aawc_county_has_authored_landmark_candidate = yes }
			}
			remove_county_modifier = aawc_rh_candidate_marker
			every_county_province = {
				limit = { has_holding = yes }
				if = { limit = { has_building = small_ruin_01 } remove_building = small_ruin_01 }
				if = { limit = { has_building = medium_ruin_01 } remove_building = medium_ruin_01 }
				if = { limit = { has_building = large_ruin_01 } remove_building = large_ruin_01 }
				if = { limit = { has_building = small_ruined_holding_01 } remove_building = small_ruined_holding_01 }
				if = { limit = { has_building = medium_ruined_holding_01 } remove_building = medium_ruined_holding_01 }
				if = { limit = { has_building = large_ruined_holding_01 } remove_building = large_ruined_holding_01 }
				if = { limit = { has_building = ruin_renowned_01 } remove_building = ruin_renowned_01 }
				if = {
					limit = { NOT = { has_holding_type = wilderness_holding } }
					set_holding_type = wilderness_holding
				}
			}
		}
	}
}

		limit = { aawc_tribal_shatter_active = yes }

		aawc_cleanup_nonlandmark_default_ruins_to_wilderness_effect = yes

		# Diagnostic: authored counties that are intentionally survivor pockets must not be
		# converted into RH ruin-holder workflow.
		every_county = {
			limit = {
				has_county_modifier = aawc_survivor_pocket_marker
				aawc_county_has_authored_landmark_candidate = yes
			}
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC FLOW convert_rh_to_lov_ruins survivor safeguard active for authored county" }
		}

		every_county = {
			limit = {
				OR = {
					AND = {
						has_county_modifier = aawc_rh_candidate_marker
						aawc_county_has_authored_landmark_candidate = yes
					}
					AND = {
						exists = title_province
						OR = {
							title_province = { has_holding_type = ruin_holding }
							title_province = { has_holding_type = wilderness_holding }
						}
						aawc_county_has_authored_landmark_candidate = yes
					}
				}
				exists = title_province
				NOT = { title_province = { has_holding_type = unknown_holding } }
				NOT = { title_province = { has_holding_type = settlement_holding } }
				NOT = { any_county_province = { has_holding_type = settlement_holding } }
				NOT = { has_county_modifier = aawc_survivor_pocket_marker }
			}

			save_scope_as = aawc_target_county
			if = { limit = { holder ?= { this = title:c_ruins.holder } } error_log = "AAWC RH CONVERT holder source: c_ruins dummy" }
			else_if = { limit = { holder ?= { this = title:c_wilderness.holder } } error_log = "AAWC RH CONVERT holder source: c_wilderness dummy" }

			# not as normal holdings under a ruin dummy.
			if = {
				limit = { NOT = { holder ?= { this = title:c_ruins.holder } } }
				hidden_effect = {
					create_title_and_vassal_change = {
						type = revoked
						save_scope_as = aawc_ruin_legitimacy_change
					}
					change_title_holder = {
						holder = title:c_ruins.holder
						change = scope:aawc_ruin_legitimacy_change
						take_baronies = yes
					}
					resolve_title_and_vassal_change = scope:aawc_ruin_legitimacy_change
					generate_coa = yes
				}
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC RH LEGITIMACY retook county for ruin holder" }
			}

				every_county_province = {
					limit = {
						has_holding = yes
						NOT = { has_holding_type = ruin_holding }
						NOT = { has_holding_type = settlement_holding }
					}
					set_holding_type = ruin_holding
				}

			if = {
				limit = { aawc_native_lov_rh_county = yes }

					if = { limit = { has_holding_type = ruin_holding } error_log = "AAWC CONVERT audit county capital ruin_holding" }
					if = { limit = { has_holding_type = wilderness_holding } error_log = "AAWC CONVERT audit county capital wilderness_holding" }

					if = { limit = { has_building = medium_ruined_holding_01 } error_log = "AAWC CONVERT audit county unexpectedly has medium_ruined_holding_01" }
