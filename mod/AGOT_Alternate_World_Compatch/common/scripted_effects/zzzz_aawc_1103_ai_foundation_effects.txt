aawc_1103_apply_ai_faith_family_the_chosen_effect = {
	random_list = {
		100 = {
			set_character_faith = faith:aawc_ai_foundation_chosen_ashen
			every_courtier = { set_character_faith = faith:aawc_ai_foundation_chosen_ashen }
			capital_county = { set_county_faith = faith:aawc_ai_foundation_chosen_ashen }
		}
		100 = {
			set_character_faith = faith:aawc_ai_foundation_chosen_dawnbound
			every_courtier = { set_character_faith = faith:aawc_ai_foundation_chosen_dawnbound }
			capital_county = { set_county_faith = faith:aawc_ai_foundation_chosen_dawnbound }
		}
		100 = {
			set_character_faith = faith:aawc_ai_foundation_chosen_veiled
			every_courtier = { set_character_faith = faith:aawc_ai_foundation_chosen_veiled }
			capital_county = { set_county_faith = faith:aawc_ai_foundation_chosen_veiled }
		}
		100 = {
			set_character_faith = faith:aawc_ai_foundation_chosen_stormbound
			every_courtier = { set_character_faith = faith:aawc_ai_foundation_chosen_stormbound }
			capital_county = { set_county_faith = faith:aawc_ai_foundation_chosen_stormbound }
		}
	}
}
