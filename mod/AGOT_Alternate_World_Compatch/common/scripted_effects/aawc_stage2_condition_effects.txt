# Stage 2 contribution API. Province scope, invoked only during structural rebuild.
# All seven parameters are required; use zero for unrelated dimensions.
# SPACE_PERCENT is physical land-use adjustment, not a prosperity bonus.
aawc_stage2_add_contribution_effect = {
	change_variable = { name = aawc_stage2_structural_sustenance add = $SUSTENANCE$ }
	change_variable = { name = aawc_stage2_structural_habitation add = $HABITATION$ }
	change_variable = { name = aawc_stage2_structural_opportunity add = $OPPORTUNITY$ }
	change_variable = { name = aawc_stage2_structural_health add = $HEALTH$ }
	change_variable = { name = aawc_stage2_structural_security add = $SECURITY$ }
	change_variable = { name = aawc_stage2_structural_amenities add = $AMENITIES$ }
	change_variable = { name = aawc_stage2_space_percent add = $SPACE_PERCENT$ }
}

aawc_stage2_refresh_effective_effect = {
	if = {
		limit = { aawc_population_valid_settlement_for_cohort_trigger = yes }
	set_variable = { name = aawc_stage2_effective_sustenance value = aawc_stage2_structural_sustenance_value }
	set_variable = { name = aawc_stage2_effective_habitation value = aawc_stage2_structural_habitation_value }
	set_variable = { name = aawc_stage2_effective_opportunity value = aawc_stage2_structural_opportunity_value }
	set_variable = { name = aawc_stage2_effective_health value = aawc_stage2_structural_health_value }
	set_variable = { name = aawc_stage2_effective_security value = aawc_stage2_structural_security_value }
	set_variable = { name = aawc_stage2_effective_amenities value = aawc_stage2_structural_amenities_value }
	set_variable = { name = aawc_stage2_occupancy_ratio value = aawc_stage2_occupancy }
	change_variable = { name = aawc_stage2_effective_habitation add = { value = aawc_stage2_housing_pressure_loss multiply = -1 } }
	change_variable = { name = aawc_stage2_effective_health add = { value = aawc_stage2_health_pressure_loss multiply = -1 } }
	# Low control affects experienced safety, not structural fortification.
	change_variable = { name = aawc_stage2_effective_security add = { value = county.county_control min = 0 max = 100 subtract = 100 multiply = 0.35 } }
	# Direct raid/siege casualties remain solely in the existing external-death API.
	# Disruption changes household safety/necessities; no repeat event death charge.
	if = {
		limit = { county = { OR = { has_variable = aawc_population_siege_growth_suppressed has_variable = aawc_population_raid_growth_suppressed } } }
	change_variable = { name = aawc_stage2_effective_security add = -30 }
	change_variable = { name = aawc_stage2_effective_sustenance add = -5 }
	}
	# The legacy epidemic growth override is dormant; this is the health input
	# for downstream age-specific ordinary mortality, not an extra population debit.
	if = {
		limit = { county = { has_province_with_epidemic = { intensity = apocalyptic } } }
	change_variable = { name = aawc_stage2_effective_health add = -45 }
	}
	else_if = {
		limit = { county = { has_province_with_epidemic = { intensity >= major } } }
	change_variable = { name = aawc_stage2_effective_health add = -30 }
	}
	else_if = {
		limit = { county = { has_province_with_epidemic = { intensity = any } } }
	change_variable = { name = aawc_stage2_effective_health add = -15 }
	}
	set_variable = { name = aawc_stage2_effective_sustenance value = { value = var:aawc_stage2_effective_sustenance min = 0 max = 100 } }
	set_variable = { name = aawc_stage2_effective_habitation value = { value = var:aawc_stage2_effective_habitation min = 0 max = 100 } }
	set_variable = { name = aawc_stage2_effective_opportunity value = { value = var:aawc_stage2_effective_opportunity min = 0 max = 100 } }
	set_variable = { name = aawc_stage2_effective_health value = { value = var:aawc_stage2_effective_health min = 0 max = 100 } }
	set_variable = { name = aawc_stage2_effective_security value = { value = var:aawc_stage2_effective_security min = 0 max = 100 } }
	set_variable = { name = aawc_stage2_effective_amenities value = { value = var:aawc_stage2_effective_amenities min = 0 max = 100 } }
	}
}

# Called by every production ledger initialization, including receipt-backed colonies.
aawc_stage2_initialize_conditions_effect = {
	aawc_stage2_refresh_structural_effect = yes
	set_variable = { name = aawc_population_expected value = aawc_stage2_physical_capacity }
	aawc_stage2_refresh_effective_effect = yes
}
