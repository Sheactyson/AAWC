# Reforging 1.11 Milestone 1 production adapters.
#
# The thirteen age bands are authoritative. aawc_population_current is retained
# only as a compatibility mirror for the established AAWC/AGOT/LoV interfaces.
# Every production population mutation enters through one of the bounded
# adapters below and finishes by reconciling bands, military overlays, the
# seven cohort mirrors and the legacy aggregate mirrors.

# This is the sole sanctioned writer of the legacy holding aggregate.
aawc_reforging_cohort_sync_legacy_population_effect = {
	set_variable = { name = aawc_population_current value = aawc_reforging_cohort_band_sum }
	set_variable = { name = aawc_population_total value = var:aawc_population_current }
	set_variable = { name = aawc_population_surplus value = var:aawc_population_current }
	if = {
		limit = { has_variable = aawc_population_expected }
		change_variable = {
			name = aawc_population_surplus
			add = { value = var:aawc_population_expected multiply = -1 }
		}
	}
}

# Native holding removal/conversion can bypass the population adapters. This
# bounded world pass repairs those paths and rebuilds mirrors only after a hit.
aawc_reforging_cohort_reconcile_world_settlement_eligibility_effect = {
	set_variable = { name = aawc_reforging_lifecycle_removed_ledgers value = 0 }
	save_scope_as = aawc_reforging_lifecycle_owner
	every_province = {
		limit = {
			NOT = { aawc_population_valid_settlement_for_cohort_trigger = yes }
			OR = {
				has_variable = aawc_reforging_cohort_production_initialized
				has_variable = aawc_reforging_cohort_age_00_04
				has_variable = aawc_population_current
				has_variable = aawc_population_preconversion_tribal
				has_variable = aawc_population_preconversion_settlement
			}
		}
		aawc_reforging_cohort_clear_ineligible_settlement_effect = yes
		scope:aawc_reforging_lifecycle_owner = { change_variable = { name = aawc_reforging_lifecycle_removed_ledgers add = 1 } }
	}
	if = {
		limit = { var:aawc_reforging_lifecycle_removed_ledgers > 0 }
		every_ruler = {
			limit = { is_landed = yes agot_is_dummy_character = no }
			aawc_population_recalculate_character_effect = yes
		}
	}
	remove_variable = aawc_reforging_lifecycle_removed_ledgers
}

# One-time bridge for holdings that predate the cohort ledger. Existing aggregate
# population is preserved exactly; no person is created or removed by migration.
aawc_reforging_cohort_ensure_production_ledger_effect = {
	if = {
		limit = { aawc_population_valid_settlement_for_cohort_trigger = yes }
	if = {
		limit = { NOT = { has_variable = aawc_reforging_cohort_production_initialized } }
		set_variable = { name = aawc_reforging_cohort_work_population value = 0 }
		if = {
			limit = { has_variable = aawc_population_current }
			set_variable = { name = aawc_reforging_cohort_work_population value = var:aawc_population_current }
		}
		aawc_reforging_diag_caller_effect = { CALLER = 7 }
		aawc_reforging_cohort_initialize_production_ledger_effect = { POPULATION = var:aawc_reforging_cohort_work_population }
		remove_variable = aawc_reforging_cohort_work_population
	}
	aawc_reforging_cohort_reconcile_effect = yes
	aawc_reforging_cohort_refresh_mirrors_effect = yes
	aawc_reforging_cohort_sync_legacy_population_effect = yes
	}
	else = { aawc_reforging_cohort_clear_ineligible_settlement_effect = yes }
}

# Guarded Milestone-1 production-candidate month. Character schedulers remain
# staggered, but title transfers can expose a holding to two scheduler phases.
# The province-local 27-day lease rejects those transition duplicates before a
# new serial is allocated. The existing serial guard still rejects repeated IDs.
aawc_reforging_cohort_apply_production_month_effect = {
	if = { limit = { OR = { NOT = { aawc_population_valid_settlement_for_cohort_trigger = yes } aawc_reforging_round_counts_valid_value = 1 } }
		if = {
			limit = { aawc_population_valid_settlement_for_cohort_trigger = yes }
		aawc_reforging_cohort_ensure_production_ledger_effect = yes
			if = {
				limit = { NOT = { has_variable = aawc_reforging_cohort_monthly_tick_recent } }
				set_variable = { name = aawc_reforging_cohort_monthly_tick_recent value = 1 days = 27 }
				change_variable = { name = aawc_reforging_cohort_requested_tick add = 1 }
				aawc_reforging_cohort_monthly_authority_effect = { TICK_ID = var:aawc_reforging_cohort_requested_tick }
			}
		aawc_reforging_cohort_sync_legacy_population_effect = yes
		}
		else = { aawc_reforging_cohort_clear_ineligible_settlement_effect = yes }
	}
	else = {
		set_variable = { name = aawc_reforging_round_invalid_rejections_total value = { value = { value = 0 if = { limit = { has_variable = aawc_reforging_round_invalid_rejections_total } add = var:aawc_reforging_round_invalid_rejections_total } } add = 1 } }
	}
}

# Internal bounded drain helper. The caller owns work_remaining.
aawc_reforging_cohort_drain_band_effect = {
	if = {
		limit = { var:aawc_reforging_cohort_work_remaining > 0 var:$BAND$ > 0 }
		set_variable = { name = aawc_reforging_cohort_work_band_debit value = var:$BAND$ }
		if = {
			limit = { var:aawc_reforging_cohort_work_band_debit > var:aawc_reforging_cohort_work_remaining }
			set_variable = { name = aawc_reforging_cohort_work_band_debit value = var:aawc_reforging_cohort_work_remaining }
		}
		change_variable = { name = $BAND$ add = { value = var:aawc_reforging_cohort_work_band_debit multiply = -1 } }
		change_variable = { name = aawc_reforging_cohort_work_remaining add = { value = var:aawc_reforging_cohort_work_band_debit multiply = -1 } }
		remove_variable = aawc_reforging_cohort_work_band_debit
	}
}

# Stage-1 transfer packets preserve the actual age bands that leave a holding.
# The migration planner remains age-blind and retains its existing destination
# selection; only the commit changes from a synthetic 20-24 credit to an exact
# debit/credit packet.  Military overlay variables are subtracted from adult
# availability so a future Stage-6 subset can never be moved as a civilian.
aawc_reforging_cohort_capture_transfer_band_effect = {
	if = {
		limit = { var:aawc_reforging_cohort_work_remaining > 0 var:$BAND$ > 0 }
		set_variable = { name = aawc_reforging_cohort_work_band_available value = var:$BAND$ }
		set_variable = { name = aawc_reforging_cohort_work_band_debit value = var:aawc_reforging_cohort_work_band_available }
		if = { limit = { var:aawc_reforging_cohort_work_band_debit > var:aawc_reforging_cohort_work_remaining } set_variable = { name = aawc_reforging_cohort_work_band_debit value = var:aawc_reforging_cohort_work_remaining } }
		change_variable = { name = $BAND$ add = { value = var:aawc_reforging_cohort_work_band_debit multiply = -1 } }
		change_variable = { name = aawc_reforging_cohort_work_remaining add = { value = var:aawc_reforging_cohort_work_band_debit multiply = -1 } }
		scope:aawc_reforging_transfer_pool_owner = { change_variable = { name = $POOL$ add = prev.var:aawc_reforging_cohort_work_band_debit } }
		remove_variable = aawc_reforging_cohort_work_band_available
		remove_variable = aawc_reforging_cohort_work_band_debit
	}
}

# Set an absolute compatibility target without bypassing cohort accounting.

		set_variable = { name = aawc_reforging_cohort_work_target value = { value = { value = 0 if = { limit = { has_variable = aawc_reforging_cohort_work_target } add = var:aawc_reforging_cohort_work_target } } floor = yes } }

	}

# Signed transfer adapter used by the existing coordinated migration commit.
# The coordinator already guarantees that connected source/destination deltas
# sum to zero; positive movers enter as young adults and negative movers debit
# the bounded source ledger.

	if = { limit = { OR = { NOT = { aawc_population_valid_settlement_for_cohort_trigger = yes } aawc_reforging_round_counts_valid_value = 1 } }
		if = {
				limit = { aawc_population_valid_settlement_for_cohort_trigger = yes }

			if = {
				limit = { var:aawc_reforging_cohort_work_signed_transfer > 0 }
				if = { limit = { has_variable = aawc_reforging_tx_migration_plan } aawc_reforging_tx_whole_credit_effect = { AMOUNT = var:aawc_reforging_cohort_work_signed_transfer } }

			}
			else_if = {
				limit = { var:aawc_reforging_cohort_work_signed_transfer < 0 }
				change_variable = { name = aawc_reforging_cohort_work_signed_transfer multiply = -1 }
				if = { limit = { has_variable = aawc_reforging_tx_migration_plan } aawc_reforging_tx_whole_debit_effect = { AMOUNT = var:aawc_reforging_cohort_work_signed_transfer } }
				else = { aawc_reforging_cohort_debit_population_effect = { AMOUNT = var:aawc_reforging_cohort_work_signed_transfer } }
			}
			else = { aawc_reforging_cohort_ensure_production_ledger_effect = yes }
			remove_variable = aawc_reforging_cohort_work_signed_transfer
			}
			else = { aawc_reforging_cohort_clear_ineligible_settlement_effect = yes }
	}
	else = {
		set_variable = { name = aawc_reforging_round_invalid_rejections_total value = { value = { value = 0 if = { limit = { has_variable = aawc_reforging_round_invalid_rejections_total } add = var:aawc_reforging_round_invalid_rejections_total } } add = 1 } }
		set_variable = { name = aawc_reforging_cohort_last_transfer_debit value = 0 }
		set_variable = { name = aawc_reforging_round_last_applied value = 0 }
	}
}
