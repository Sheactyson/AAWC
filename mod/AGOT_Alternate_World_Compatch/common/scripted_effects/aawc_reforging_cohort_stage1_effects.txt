# Reforging 1.11 Milestone 1 cohort engine.
#
# These are the bounded cohort-ledger primitives. Production adapters reuse
# their credit, debit, death, reconciliation, and mirror-refresh operations,
# while the guarded recurring demographic tick is reached only through the
# single audited production adapter. Its candidate rates remain defined in
# aawc_reforging_cohort_stage1_values.txt; opt-in diagnostics stay isolated in
# the Stage-1 test event files.

# Rebuild the seven stored UI/compatibility mirrors from authoritative bands.
# This effect never changes an age band, military overlay, or fractional carry.
aawc_reforging_cohort_refresh_mirrors_effect = {
	set_variable = { name = aawc_reforging_cohort_children_mirror value = aawc_reforging_cohort_children }
	set_variable = { name = aawc_reforging_cohort_young_adults_mirror value = aawc_reforging_cohort_young_adults }
	set_variable = { name = aawc_reforging_cohort_older_adults_mirror value = aawc_reforging_cohort_older_adults }
	set_variable = { name = aawc_reforging_cohort_elderly_mirror value = aawc_reforging_cohort_elderly }

	set_variable = { name = aawc_reforging_military_total_mirror value = aawc_reforging_military_overlay_sum }
	set_variable = { name = aawc_reforging_civilian_adults_mirror value = aawc_reforging_civilian_available_adults }
}

# Consume the caller's whole-person building-authored training budget from one
# adult band, bounded by that band's explicit building eligibility. This is
# support classification only: no regiment identity, station, soldier count or
# casualty cause is created or inferred.
aawc_reforging_cohort_train_military_band_effect = {
	if = {
		limit = {
			var:aawc_reforging_military_training_work_remaining > 0
			var:$BAND$ > var:$MILITARY$
			$ELIGIBILITY$ > var:$MILITARY$
		}
		set_variable = { name = aawc_reforging_military_training_band_room value = $ELIGIBILITY$ }
		if = {
			limit = { var:aawc_reforging_military_training_band_room > var:$BAND$ }
			set_variable = { name = aawc_reforging_military_training_band_room value = var:$BAND$ }
		}
		change_variable = { name = aawc_reforging_military_training_band_room add = { value = var:$MILITARY$ multiply = -1 } }
		if = {
			limit = { var:aawc_reforging_military_training_band_room > 0 }
			set_variable = { name = aawc_reforging_military_training_band_amount value = var:aawc_reforging_military_training_work_remaining }
			if = {
				limit = { var:aawc_reforging_military_training_band_amount > var:aawc_reforging_military_training_band_room }
				set_variable = { name = aawc_reforging_military_training_band_amount value = var:aawc_reforging_military_training_band_room }
			}
			aawc_reforging_cohort_classify_military_effect = { BAND = $BAND$ MILITARY = $MILITARY$ AMOUNT = var:aawc_reforging_military_training_band_amount }
			change_variable = { name = aawc_reforging_military_training_work_remaining add = { value = aawc_reforging_round_last_applied_value multiply = -1 } }
			change_variable = { name = aawc_reforging_cohort_last_military_trained add = aawc_reforging_round_last_applied_value }
		}
		remove_variable = aawc_reforging_military_training_band_room
		remove_variable = aawc_reforging_military_training_band_amount
	}
}

# Building-driven support-training interface. The building layer owns both the
# age-specific eligibility ceilings and monthly throughput. The current
# review-only catalogue supplies neither, so every contract value is zero and
# this effect preserves existing military overlays without automatically
# recruiting anyone. A later authorized adapter may supply fractional
# throughput; any whole demand above current per-age eligibility is discarded
# rather than becoming a hidden backlog after a later building change.
aawc_reforging_cohort_apply_military_training_effect = {
	set_variable = { name = aawc_reforging_cohort_last_military_trained value = 0 }
	if = { limit = { NOT = { has_variable = aawc_reforging_military_training_fraction } } set_variable = { name = aawc_reforging_military_training_fraction value = 0 } }
	if = {
		limit = {
			aawc_reforging_civilian_available_adults > 0
			aawc_reforging_military_building_training_monthly > 0
		}
		change_variable = { name = aawc_reforging_military_training_fraction add = aawc_reforging_military_building_training_monthly }
		if = {
			limit = { var:aawc_reforging_military_training_fraction >= 1 }
			set_variable = { name = aawc_reforging_military_training_work_remaining value = { value = var:aawc_reforging_military_training_fraction floor = yes } }
			change_variable = { name = aawc_reforging_military_training_fraction add = { value = var:aawc_reforging_military_training_work_remaining multiply = -1 } }
			aawc_reforging_cohort_train_military_band_effect = { BAND = aawc_reforging_cohort_age_15_19 MILITARY = aawc_reforging_military_age_15_19 ELIGIBILITY = aawc_reforging_military_building_eligibility_age_15_19 }
			aawc_reforging_cohort_train_military_band_effect = { BAND = aawc_reforging_cohort_age_20_24 MILITARY = aawc_reforging_military_age_20_24 ELIGIBILITY = aawc_reforging_military_building_eligibility_age_20_24 }

			aawc_reforging_cohort_train_military_band_effect = { BAND = aawc_reforging_cohort_age_30_34 MILITARY = aawc_reforging_military_age_30_34 ELIGIBILITY = aawc_reforging_military_building_eligibility_age_30_34 }
			aawc_reforging_cohort_train_military_band_effect = { BAND = aawc_reforging_cohort_age_35_39 MILITARY = aawc_reforging_military_age_35_39 ELIGIBILITY = aawc_reforging_military_building_eligibility_age_35_39 }

# Sole Stage-1 demographic-tick authority. TICK_ID is caller-supplied and a
# repeated ID is a no-op. A re-entrant call is also a no-op. The guarded
# production candidate reaches this authority only through the audited adapter
# in aawc_reforging_cohort_production_effects.txt; the explicit test adapter
# remains available for deterministic and rendered regression work.
aawc_reforging_cohort_monthly_authority_effect = {
	if = { limit = { aawc_reforging_round_counts_valid_value = 1 }
		if = {
			limit = {
				has_variable = aawc_reforging_cohort_stage1_initialized
				NOT = { has_variable = aawc_reforging_cohort_tick_in_progress }
				NOT = { var:aawc_reforging_cohort_last_tick = $TICK_ID$ }
			}
			aawc_reforging_diag_check_effect = { STAGE = 1 }
			set_variable = { name = aawc_reforging_cohort_tick_in_progress value = 1 }
			set_variable = { name = aawc_reforging_cohort_last_before_total value = aawc_reforging_cohort_band_sum }
			set_variable = { name = aawc_reforging_cohort_last_births value = 0 }
			set_variable = { name = aawc_reforging_cohort_last_deaths value = 0 }
			set_variable = { name = aawc_reforging_cohort_last_aged value = 0 }
			set_variable = { name = aawc_reforging_cohort_last_military_aged value = 0 }
			set_variable = { name = aawc_reforging_cohort_last_military_retired value = 0 }
	
			aawc_reforging_cohort_apply_births_effect = yes
			aawc_reforging_cohort_apply_mortality_effect = yes
			aawc_reforging_cohort_apply_aging_effect = yes
			aawc_reforging_cohort_apply_military_training_effect = yes
			aawc_reforging_cohort_reconcile_effect = yes
			aawc_reforging_cohort_refresh_mirrors_effect = yes
	
			set_variable = { name = aawc_reforging_cohort_last_after_total value = aawc_reforging_cohort_band_sum }
			set_variable = { name = aawc_reforging_cohort_last_expected_after value = var:aawc_reforging_cohort_last_before_total }
			change_variable = { name = aawc_reforging_cohort_last_expected_after add = var:aawc_reforging_cohort_last_births }
			change_variable = { name = aawc_reforging_cohort_last_expected_after add = { value = var:aawc_reforging_cohort_last_deaths multiply = -1 } }
			set_variable = { name = aawc_reforging_cohort_last_tick value = $TICK_ID$ }
			change_variable = { name = aawc_reforging_cohort_tick_count add = 1 }
			remove_variable = aawc_reforging_cohort_tick_in_progress
			aawc_reforging_diag_check_effect = { STAGE = 2 }
		}
	}
	else = {
		set_variable = { name = aawc_reforging_round_invalid_rejections_total value = { value = { value = 0 if = { limit = { has_variable = aawc_reforging_round_invalid_rejections_total } add = var:aawc_reforging_round_invalid_rejections_total } } add = 1 } }
	}
}

# Bounded CK3-native controls for external wall-clock comparison. Both routes
# scan the same world and cap at 1,024 landed provinces x 12 inner operations.
# The cohort route uses transient isolated ledgers and removes every field.
aawc_reforging_cohort_stage1_benchmark_baseline_effect = {
	set_variable = { name = aawc_reforging_benchmark_count value = 0 }
	set_variable = { name = aawc_reforging_benchmark_operations value = 0 }
	set_variable = { name = aawc_reforging_benchmark_population_checksum value = 0 }
	set_variable = { name = aawc_reforging_benchmark_military_checksum value = 0 }
	set_variable = { name = aawc_reforging_benchmark_invariant_failures value = 0 }
	save_scope_as = aawc_reforging_benchmark_owner
	every_province = {
		limit = {
			exists = county
			NOT = { has_variable = aawc_reforging_cohort_test_initialized }
		}
		if = {
			limit = {
				exists = scope:aawc_reforging_benchmark_owner
				scope:aawc_reforging_benchmark_owner = {
					has_variable = aawc_reforging_benchmark_count
					var:aawc_reforging_benchmark_count < 1024
				}
			}
			scope:aawc_reforging_benchmark_owner = {
				change_variable = { name = aawc_reforging_benchmark_count add = 1 }
			}
			while = {
				count = 12
				scope:aawc_reforging_benchmark_owner = {
					change_variable = { name = aawc_reforging_benchmark_operations add = 1 }
				}
			}
		}
	}
	debug_log = "AAWC REFORGING STAGE1 benchmark baseline complete"
}
