# Diagnostic-only evidence. Summary/Verbose enable it; Off adds no saved state.
# Never resets, rounds, relocates or repairs any resident, cohort or carry.
# Sequence markers order synchronous calls, not assumptions about on_action ordering.
aawc_reforging_diag_phase_effect = {
    if = {
        limit = { aawc_diagnostics_summary_enabled = yes }
        if = { limit = { NOT = { has_global_variable = aawc_reforging_diag_sequence } } set_global_variable = { name = aawc_reforging_diag_sequence value = 0 } }
        change_global_variable = { name = aawc_reforging_diag_sequence add = 1 }
        if = {
            limit = { NOT = { has_global_variable = aawc_reforging_diag_phase_$PHASE$_visits } }
            set_global_variable = { name = aawc_reforging_diag_phase_$PHASE$_visits value = 0 }
            set_global_variable = { name = aawc_reforging_diag_phase_$PHASE$_first_sequence value = global_var:aawc_reforging_diag_sequence }
        }
        change_global_variable = { name = aawc_reforging_diag_phase_$PHASE$_visits add = 1 }
        set_global_variable = { name = aawc_reforging_diag_phase value = $PHASE$ }
        set_global_variable = { name = aawc_reforging_diag_phase_$PHASE$_sequence value = global_var:aawc_reforging_diag_sequence }
        set_global_variable = { name = aawc_reforging_diag_phase_$PHASE$_date value = current_date }
        debug_log = "AAWC DIAG phase marker date=[GetCurrentDate.GetStringLong]"
    }
}

# One bounded world heartbeat per twelve successful migration-event entries.
# Saved count/date advance even when native script logging stops; no world scan.
aawc_reforging_diag_heartbeat_effect = {
    if = {
        limit = { aawc_diagnostics_summary_enabled = yes }
        if = {
            limit = { NOT = { has_global_variable = aawc_reforging_diag_world_pulses } }
            set_global_variable = { name = aawc_reforging_diag_world_pulses value = 0 }
            set_global_variable = { name = aawc_reforging_diag_world_since_log value = 11 }
            set_global_variable = { name = aawc_reforging_diag_world_heartbeats value = 0 }
        }
        change_global_variable = { name = aawc_reforging_diag_world_pulses add = 1 }
        change_global_variable = { name = aawc_reforging_diag_world_since_log add = 1 }
        set_global_variable = { name = aawc_reforging_diag_world_last_date value = current_date }
        if = {
            limit = { global_var:aawc_reforging_diag_world_since_log >= 12 }
            set_global_variable = { name = aawc_reforging_diag_world_since_log value = 0 }
            change_global_variable = { name = aawc_reforging_diag_world_heartbeats add = 1 }
            # Native debug emission can stop after sustained output while the
            # error/game route remains live. Mirror this sparse record; coverage
            # counts the largest single channel and never sums the two copies.
            error_log = "AAWC DIAG world heartbeat date=[GetCurrentDate.GetStringLong]"
            debug_log = "AAWC DIAG world heartbeat date=[GetCurrentDate.GetStringLong]"
        }
    }
}
