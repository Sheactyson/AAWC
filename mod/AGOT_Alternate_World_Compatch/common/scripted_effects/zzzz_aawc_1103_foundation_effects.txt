# Foundation setup. Player foundation events are interactive; AI foundations
# run automatically in the same delayed startup phase.
aawc_1103_invite_player_foundations_effect = {
	if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC 1.10.4.6 player foundation invitation scan begin" }
	every_player = {
		limit = {
			OR = {
				is_landed = yes
				AND = {
					has_game_rule = aawc_age_of_silence
					is_landless_adventurer = yes
				}
			}
			agot_is_dummy_character = no
			OR = {
				has_game_rule = aawc_lost_faith_culture_enabled
				has_game_rule = aawc_lost_faith_culture_faith_only
				has_game_rule = aawc_lost_faith_culture_culture_only
			}
		}
		if = {
			limit = {
				NOT = { has_character_flag = aawc_1103_culture_foundation_invited }
				OR = {
					has_game_rule = aawc_lost_faith_culture_enabled
					has_game_rule = aawc_lost_faith_culture_culture_only
				}
			}
			trigger_event = { id = aawc_foundation.0001 days = 3 }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC 1.10.4.6 player culture foundation invitation queued after landed-state stabilization" }
		}
		if = {
			limit = {
				NOT = { has_character_flag = aawc_1103_faith_foundation_invited }
				OR = {
					has_game_rule = aawc_lost_faith_culture_enabled
					has_game_rule = aawc_lost_faith_culture_faith_only
				}
			}
			trigger_event = { id = aawc_foundation.0002 days = 4 }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC 1.10.4.6 player faith foundation invitation queued after landed-state stabilization" }
		}
	}
}

# Culture divergence can already leave the selected pillar on the new culture.
# Reapplying the identical pillar is a CK3 script error, so make the finalizer
# idempotent without changing the selected foundation output.
aawc_1103_set_foundation_pillar_if_needed_effect = {
	if = {
		limit = { NOT = { has_cultural_pillar = $PILLAR$ } }
		set_culture_pillar = $PILLAR$
	}
}

aawc_1103_finalize_player_culture_foundation_effect = {
	# Restore the player-proven 1.10.3.8 flow. Creation converts this event
	# root synchronously, so root.culture is the new divergence immediately.
	save_scope_as = aawc_1103_founder
	if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC 1.10.3 direct founder culture finalizer started" }
	aawc_1103_prepare_player_culture_visual_seed_effect = yes
	# Normalize the temporary source before divergence. This prevents inherited
	# history from establishing a later era or carrying extra innovations forward.
	culture = {
		aawc_1102_reset_lost_culture_innovations_effect = yes
		aawc_1103_clear_foundation_traditions_effect = yes
	}
	set_global_variable = { name = aawc_1103_culture_namer value = root }
	create_divergent_culture_with_side_effects_excluding_cost = yes
	culture = {
		aawc_1103_finalize_created_foundation_culture_effect = yes
	}
}
