			}
			domicile = {
				domicile_location = { county = scope:wilderness.county }
			}
			custom_tooltip = {
				text = aawc_age_of_silence_can_settle_wilderness_tt
				scope:wilderness = {
					province_owner = { agot_is_dummy_character = yes }
					has_holding_type = wilderness_holding
				}
			}
			NOR = {
				var:lv_settlement_state ?= flag:planning
				any_owned_story = {
					story_type = lv_settlement_story_cycle
				}
			}
		}
		trigger_else = {

		NOT = { government_has_flag = government_is_nw }
		NOT = { government_has_flag = government_is_uninteractable }
		custom_tooltip = {
			text = aawc_population_top_liege_only_tt
			aawc_population_is_top_liege_actor = yes
		}

		trigger_if = {
			limit = {
				OR = {
					has_game_rule = lv_colonisation_agot
					scope:wilderness.county = { aawc_county_should_use_agot_colonization_flow = yes }
				}
			}
			scope:wilderness.county = {
				custom_tooltip = {
					text = aawc_population_need_colonize_surplus_wilderness_gui_tt
					root.var:aawc_population_available >= aawc_population_county_colonize_cost_wh
				}
			}
			custom_tooltip = {
				text = aawc_population_colonize_gold_cost_summary_tt
				gold >= colonize_cost_val
			}
			trigger_if = {
				limit = {
					scope:wilderness ?= { geographical_region = world_westeros_beyond_the_wall }
					culture = { agot_is_wildling_culture = no }
					NOT = {
						any_sub_realm_county = {
							any_neighboring_county = {
								this = scope:wilderness.county
							}
						}
					}
				}
				custom_tooltip = {
					text = colonize_btw
					always = no
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = colonize_border_county_trigger_tooltip
					# Only surrounding lands
					any_sub_realm_county = {
						any_title_to_title_neighboring_and_across_water_county = {
							this = scope:wilderness.county
						}
					}
				}
			}
			any_sub_realm_county = {
				any_title_to_title_neighboring_and_across_water_county = {
					NOT = { has_county_modifier = block_settlement_ability }
				}
			}
			custom_tooltip = {
				text = colonization_cooldown
				NOT = { has_variable = colonization_cooldown }
			}
		}
		trigger_else = {
			scope:wilderness = { save_temporary_scope_as = ruin }
			scope:wilderness.county = {
				custom_tooltip = {
					text = aawc_population_need_colonize_surplus_wilderness_gui_tt
					root.var:aawc_population_available >= aawc_population_county_colonize_cost_wh
				}
			}
			custom_tooltip = {
				text = aawc_population_colonize_gold_cost_summary_tt
				gold >= lv_settlement_cost
			}
			NOR = {
				var:lv_settlement_state ?= flag:planning
				any_owned_story = {
					story_type = lv_settlement_story_cycle
					is_target_in_variable_list = {
						name = lv_settlement_expeditions
						target = scope:ruin
					}
				}
			}
		}
		}
	}

	## Check agot_colonization_effects.txt
	effect = {
		if = {
			limit = {
				has_game_rule = aawc_age_of_silence
				is_landless_adventurer = yes
				has_realm_law = aawc_camp_purpose_colonization
			}
			custom_tooltip = aawc_age_of_silence_settle_wilderness_effect_tt
		}
		else = {
			scope:wilderness.county = { custom_tooltip = aawc_population_colonize_surplus_cost_desc }
			custom_tooltip = colonize_land_cost_desc
			custom_tooltip = colonize_land_holding_desc
		}
		hidden_effect = {
			if = {
				limit = {
					has_game_rule = aawc_age_of_silence
					root = {
						is_landless_adventurer = yes
						has_realm_law = aawc_camp_purpose_colonization
					}
				}
				scope:wilderness = { save_scope_as = ruin }
				root = {
					set_variable = {
						name = lv_restore_ruins_tooltip
						value = scope:ruin
					}
					save_scope_value_as = { name = lv_settlement_cost value = 0 }
					save_scope_value_as = { name = aawc_population_settlement_cost value = 0 }
					error_log = "AAWC 1.10.4 Age of Silence starting zero-cost wilderness expedition"
					trigger_event = lv_settlement_event.0001
				}
			}
			else_if = {
				limit = {
					OR = {
						has_game_rule = lv_colonisation_agot
						scope:wilderness.county = { aawc_county_should_use_agot_colonization_flow = yes }
					}
				}
				root = {
					aawc_population_recalculate_character_effect = yes
					if = {
						limit = {
							aawc_population_is_top_liege_actor = yes
							var:aawc_population_available >= scope:wilderness.county.aawc_population_county_colonize_cost_wh
							gold >= colonize_cost_val
						}
						save_scope_value_as = {
							name = aawc_population_settlement_cost
							value = scope:wilderness.county.aawc_population_county_colonize_cost_wh
						}
						aawc_population_add_reserved_amount_effect = {
							COST = scope:aawc_population_settlement_cost
						}
						scope:wilderness = { save_scope_as = aawc_population_paid_wilderness }
					}
					else = {
						error_log = "AAWC POP AGOT colonization blocked at execution because the actor lacked the full gold or population cost"
					}
				}
				if = {
					limit = { exists = scope:aawc_population_paid_wilderness }
					ai_colonization_effect = { WILDERNESS = scope:wilderness ROOT_SCOPE = root }
					root = {
						aawc_population_consume_reserved_amount_effect = {
							COST = scope:aawc_population_settlement_cost
						}
						aawc_population_prepare_settlement_receipt_effect = { STARTING_POPULATION = 100 }
						remove_variable = aawc_population_settlement_cost
						error_log = "AAWC POP AGOT colonize consumed the full reserved population cost"
					}
					scope:aawc_population_paid_wilderness = {
						if = { limit = { aawc_population_valid_settlement_for_cohort_trigger = yes } aawc_reforging_cohort_initialize_production_ledger_effect = { POPULATION = root.var:aawc_population_last_settlement_credit } }
					}
					root = { remove_variable = aawc_population_pending_settlement_credit }
				}
			}
			else = {
				scope:wilderness = { save_scope_as = ruin }
				error_log = "AAWC DEBUG agot_can_colonize LoV expedition branch"
				root = {
					set_variable = {
						name = lv_restore_ruins_tooltip
						value = scope:ruin
					}
					error_log = "AAWC DEBUG agot_can_colonize set lv_restore_ruins_tooltip"
					if = {
						limit = {
							aawc_population_is_top_liege_actor = yes
							var:aawc_population_available >= scope:wilderness.county.aawc_population_county_colonize_cost_wh
							gold >= lv_settlement_cost
						}
						save_scope_value_as = {
							name = lv_settlement_cost
							value = lv_settlement_cost
						}
						save_scope_value_as = {
							name = aawc_population_settlement_cost
							value = scope:wilderness.county.aawc_population_county_colonize_cost_wh
						}
						error_log = "AAWC DEBUG agot_can_colonize triggering lv_settlement_event.0001"
						trigger_event = lv_settlement_event.0001
					}
					else = {
						error_log = "AAWC DEBUG agot_can_colonize blocked by gold, top-liege, or population surplus requirements"
					}
				}
			}
		}
	}
}
