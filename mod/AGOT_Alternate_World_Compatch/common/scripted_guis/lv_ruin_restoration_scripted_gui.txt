# AAWC ownership/state predicate for county-view wilderness actions.
# AGOT's stock predicate only checks the holder's wilderness trait.  That trait
# can legitimately be replaced by another dummy trait when technical dummy
# titles share a holder, as happened to the Wilderness ruler in older saves.
aawc_is_colonizable_wilderness = {
	scope = province
	is_shown = {
		exists = root
		is_county_capital = yes
		has_holding_type = wilderness_holding
		county = {
			NOR = {
				this = title:c_unknown
				this = title:c_ruins
				this = title:c_wilderness
			}
			OR = {
				holder ?= { agot_is_dummy_character = yes }
				holder ?= { this = title:c_wilderness.holder }
			}
		}
	}
}

# Used for restoration cost calculation
lv_restore_ruins_tooltip_show = {
	scope = province
	effect = {
		aawc_population_cache_selected_holding_levy_debug_effect = yes
		save_scope_as = ruin
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC DEBUG lv_restore_ruins_tooltip_show invoked" }
		scope:actor = {
			# Tooltip display needs current aggregates, not another deep realm/formula scan.
			aawc_population_reaggregate_character_cached_effect = yes
			set_variable = {
				name = lv_restore_ruins_tooltip
				value = scope:ruin
			}
			set_variable = {
				name = aawc_population_tooltip_colonize_cost_wh
				value = scope:ruin.county.aawc_population_county_colonize_cost_wh
			}
			set_variable = {
				name = aawc_population_tooltip_reclaim_cost_rh
				value = scope:ruin.county.aawc_population_county_reclaim_cost_rh
			}
			set_variable = {
				name = aawc_population_tooltip_gold_cost
				value = lv_settlement_cost
			}
			set_variable = {
				name = aawc_population_available_display
				value = var:aawc_population_available
			}
			set_variable = {
				name = aawc_population_tooltip_colonize_missing_wh
				value = var:aawc_population_tooltip_colonize_cost_wh
			}
			change_variable = {
				name = aawc_population_tooltip_colonize_missing_wh
				add = { value = var:aawc_population_available multiply = -1 }
			}
			if = {
				limit = { var:aawc_population_tooltip_colonize_missing_wh < 0 }
				set_variable = { name = aawc_population_tooltip_colonize_missing_wh value = 0 }
			}
			set_variable = {
				name = aawc_population_tooltip_reclaim_missing_rh
				value = var:aawc_population_tooltip_reclaim_cost_rh
			}
			change_variable = {
				name = aawc_population_tooltip_reclaim_missing_rh
				add = { value = var:aawc_population_available multiply = -1 }
			}
			if = {
				limit = { var:aawc_population_tooltip_reclaim_missing_rh < 0 }
				set_variable = { name = aawc_population_tooltip_reclaim_missing_rh value = 0 }
			}
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC DEBUG lv_restore_ruins_tooltip set on actor" }
		}
	}
}
lv_restore_ruins_tooltip_hide = {
	scope = province
	effect = { scope:actor = { remove_variable = lv_restore_ruins_tooltip } }
}

aawc_begin_lov_expedition_from_gui = {
	scope = province

	is_valid = {
		exists = root
		NOT = { has_holding_type = wilderness_holding }
		scope:actor = {
			custom_tooltip = {
				text = aawc_population_reclaim_cost_summary_tt
				gold >= lv_settlement_cost
			}
			custom_tooltip = {
				text = aawc_population_need_reclaim_surplus_interaction_tt
				var:aawc_population_available >= root.county.aawc_population_county_reclaim_cost_rh
			}
			custom_tooltip = {
				text = aawc_population_top_liege_only_tt
				aawc_population_is_top_liege_actor = yes
			}
			custom_tooltip = {
				text = aawc_population_not_planning_tt
				NOT = { var:lv_settlement_state ?= flag:planning }
			}
			custom_tooltip = {
				text = aawc_population_target_not_already_targeted_tt
				NOT = {
					any_owned_story = {
						story_type = lv_settlement_story_cycle
						is_target_in_variable_list = {
							name = lv_settlement_expeditions
							target = root
						}
					}
				}
			}
		}
	}

	effect = {
		root = { save_scope_as = ruin }
		scope:actor = {
			aawc_population_recalculate_character_effect = yes
			set_variable = {
				name = lv_restore_ruins_tooltip
				value = scope:ruin
			}
			set_variable = {
				name = aawc_population_tooltip_reclaim_cost_rh
				value = scope:ruin.county.aawc_population_county_reclaim_cost_rh
			}
			set_variable = {
				name = aawc_population_tooltip_gold_cost
				value = lv_settlement_cost
			}
			set_variable = {
				name = aawc_population_available_display
				value = var:aawc_population_available
			}
			if = {
				limit = {
					aawc_population_is_top_liege_actor = yes
					scope:ruin = { NOT = { has_holding_type = wilderness_holding } }
					NOT = { var:lv_settlement_state ?= flag:planning }
					NOT = {
						any_owned_story = {
							story_type = lv_settlement_story_cycle
							is_target_in_variable_list = {
								name = lv_settlement_expeditions
								target = scope:ruin
							}
						}
					}
					gold >= lv_settlement_cost
					var:aawc_population_available >= scope:ruin.county.aawc_population_county_reclaim_cost_rh
				}
				save_scope_value_as = {
					name = lv_settlement_cost
					value = lv_settlement_cost
				}
				save_scope_value_as = {
					name = aawc_population_settlement_cost
					value = scope:ruin.county.aawc_population_county_reclaim_cost_rh
				}
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC DEBUG aawc_begin_lov_expedition_from_gui triggering lv_settlement_event.0001" }
				trigger_event = lv_settlement_event.0001
			}
			else = {
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC DEBUG aawc_begin_lov_expedition_from_gui blocked by resource or planning requirements" }
			}
		}
	}
}

	effect = {
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC OWNED SETTLEMENT 1.9.70 begin establish from county GUI" }
		root = { save_scope_as = aawc_owned_settlement_province }

			aawc_population_recalculate_character_effect = yes
			save_scope_value_as = {
				name = aawc_population_owned_settlement_population_cost

			}
			save_scope_value_as = {
				name = aawc_population_owned_settlement_gold_cost
				value = scope:aawc_owned_settlement_province.aawc_population_owned_settlement_gold_cost
			}
			save_scope_value_as = {
				name = aawc_population_owned_settlement_starting_population
				value = scope:aawc_owned_settlement_province.aawc_population_owned_settlement_starting_population
			}
			set_variable = {
				name = aawc_population_tooltip_owned_settlement_cost
				value = scope:aawc_population_owned_settlement_population_cost
			}
			set_variable = {
				name = aawc_population_available_display
				value = var:aawc_population_available
			}
			if = {
					limit = {
						aawc_population_is_top_liege_actor = yes
						gold >= scope:aawc_population_owned_settlement_gold_cost

					scope:aawc_owned_settlement_province = {
						has_holding = no
						is_county_capital = no
						county = { holder = scope:actor }
					}
				}
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC OWNED SETTLEMENT 1.9.70 validation passed" }
				remove_short_term_gold = scope:aawc_population_owned_settlement_gold_cost
				aawc_population_add_reserved_amount_effect = { COST = scope:aawc_population_owned_settlement_population_cost }
				aawc_population_consume_reserved_amount_effect = { COST = scope:aawc_population_owned_settlement_population_cost }
				aawc_population_prepare_settlement_receipt_effect = { STARTING_POPULATION = scope:aawc_population_owned_settlement_starting_population }

					set_holding_type = settlement_holding

			custom_tooltip = {
				text = blockers_remain_in_settlement

					flag = obstacle

				}
			}
			custom_tooltip = {

				county = { development_level >= 3 }
			}

				text = has_fully_upgraded_settlement_holding

			}
			custom_tooltip = {
				text = aawc_population_feudalize_no_construction_tt
				has_ongoing_construction = no
			}

			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC OWNED SETTLEMENT 1.9.91 begin non-capital settlement tribalize from county GUI" }
			root = { save_scope_as = aawc_tribalize_owned_settlement_province }
			scope:actor = {
				if = {
					limit = {
						aawc_population_is_top_liege_actor = yes
						government_has_flag = government_is_tribal

							has_holding_type = settlement_holding
							is_county_capital = no
							county = { holder = scope:actor }
							has_building_with_flag = {
								flag = obstacle
								count = 0
							}
							county = { development_level >= 3 }
							has_building = settlement_03
							has_ongoing_construction = no
						}
					}
					scope:aawc_tribalize_owned_settlement_province = {
						if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC OWNED SETTLEMENT 1.9.91 non-capital settlement converting to tribal_holding from tribal government" }

						aawc_population_refresh_holding_dynamic_effect = yes
						county = { aawc_population_recalculate_county_effect = yes }
					}
					if = {
						limit = {
							has_variable = num_settled_wilderness
							var:num_settled_wilderness > 0
						}
						decrease_variable = {
							NAME = num_settled_wilderness
							AMOUNT = 1
						}
						remove_wilderness_tracker_variable = yes
						if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC OWNED SETTLEMENT 1.9.91 decremented num_settled_wilderness after non-capital tribalize" }
					}
					aawc_population_resync_settlement_tracker_effect = yes
					aawc_population_recalculate_character_effect = yes
				}
				else = {
					if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC OWNED SETTLEMENT 1.9.91 blocked during non-capital tribalize effect validation" }
				}
			}
		}
	}
