aawc_can_create_castle_holding = {
	scope = province
	saved_scopes = { actor }

	is_valid = {
		exists = root
		scope:actor = {
			top_liege = {
				custom_tooltip = {
					text = aawc_population_need_new_castle_holding_surplus_tt
					var:aawc_population_available >= aawc_population_new_castle_holding_cost
				}
			}
		}
	}
}
