	}
}

# Mercenary companies are created by the engine from culture county counts.
# Validate the title's recorded capital after one day, when CK3's native
# mercenary manager has finished assigning the source county and company title.
aawc_startup.0003 = {
	type = character_event
	hidden = yes
	trigger = { aawc_tribal_shatter_active = yes }
	immediate = {
		aawc_lineage_initialize_counters_effect = yes
		if = {
			limit = {
				primary_title ?= { is_mercenary_company = yes }
				primary_title.title_capital_county ?= {
					aawc_mercenary_settled_county_trigger = yes
				}
			}
			add_character_flag = aawc_lineage_known_mercenary
			add_character_flag = aawc_lineage_audited
			every_courtier = {
				limit = {
					is_lowborn = yes
					NOT = { exists = father }
					NOT = { exists = mother }
					NOT = { has_character_flag = aawc_lineage_known_birth }
					NOT = { has_character_flag = aawc_1101_generated_start_courtier }
					NOT = { any_player = { this = prev } }
				}
				change_global_variable = { name = aawc_lineage_valid_mercenary_entourage_removed add = 1 }
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE removed engine-generated entourage from valid mercenary company" }
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
				death = { death_reason = death_disappearance }
			}
			every_courtier = {
				add_character_flag = aawc_lineage_known_mercenary
				add_character_flag = aawc_lineage_audited
			}
			change_global_variable = { name = aawc_lineage_mercenaries_accepted add = 1 }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE mercenary accepted: title capital is a settled non-dummy source county" }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
		}
		else_if = {
			limit = { primary_title ?= { is_mercenary_company = yes } }
			save_scope_as = aawc_invalid_mercenary_leader
			primary_title = { save_scope_as = aawc_invalid_mercenary_title }
			change_global_variable = { name = aawc_lineage_mercenaries_rejected add = 1 }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE mercenary rejected: title capital is not a settled non-dummy source county" }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
			every_courtier = {
				limit = {
					agot_is_dummy_character = no
					NOT = { any_player = { this = prev } }
				}
				change_global_variable = { name = aawc_lineage_invalid_mercenary_entourage_removed add = 1 }
				death = { death_reason = death_disappearance }
			}
			destroy_title = scope:aawc_invalid_mercenary_title
			death = { death_reason = death_disappearance }
		}
	}
}

# The one-time baseline removes setup pool bodies. CK3 recreates its stable
# reserve immediately afterward, so identify those replacements on a short
# delay without deleting or replacing them again.
aawc_startup.0006 = {
	type = character_event
	hidden = yes
	trigger = { aawc_tribal_shatter_active = yes }
	immediate = {
		every_living_character = {
			limit = { aawc_is_engine_pool_reserve_candidate = yes }
			aawc_lineage_quarantine_engine_reserve_effect = yes
		}
	}
}

# AI rulers receive one month to staff their government from AAWC's marked
# noble pool. Core councillors and a diarch remain; unused nobles and holders
# of optional court positions are removed. Any parentless lowborn service
# character generated during the window is removed as well.
aawc_startup.0007 = {
	type = character_event
	hidden = yes
	trigger = { aawc_tribal_shatter_active = yes }
	immediate = {
		if = {
			limit = {
				is_ai = no
				is_landed = yes
				agot_is_dummy_character = no
				has_character_flag = aawc_lineage_ai_staffing_window
			}
			aawc_1104_normalize_selected_player_staffing_pool_effect = yes
		}
		if = {
			limit = {
				is_ai = yes
				is_landed = yes
				agot_is_dummy_character = no
				has_character_flag = aawc_lineage_ai_staffing_window
			}
			save_scope_as = aawc_lineage_staffing_ruler
			every_courtier_or_guest = {
				limit = { has_character_flag = aawc_lineage_ai_staffing_candidate }
				if = {
					limit = {
						OR = {
							is_councillor = yes
							is_diarch = yes
						}
					}
					remove_character_flag = aawc_lineage_ai_staffing_candidate
					add_character_flag = aawc_lineage_ai_staffing_retained
					if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE retained marked AI staffing noble in an essential government role" }
					if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
				}
				else = {
					if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE removed unused or optional-role AI staffing noble after 30-day window" }
					if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
					death = { death_reason = death_disappearance }
				}
			}
			# Realm priests and some other generated service holders are employed
			# by the ruler without being exposed by every_courtier_or_guest. Scan
			# living characters by employer so those optional lowborns do not
			# survive merely because CK3 placed them in a service scope.
			every_living_character = {
				limit = {
					is_landed = no
					is_lowborn = yes
					exists = employer
					employer = scope:aawc_lineage_staffing_ruler
					NOT = { exists = father }
					NOT = { exists = mother }
					NOT = { has_character_flag = aawc_lineage_known_birth }
					NOT = { has_character_flag = aawc_lineage_known_aawc_generation }
					NOT = { has_character_flag = aawc_engine_pool_reserve }
					NOT = { has_character_flag = aawc_lineage_known_mercenary }
					NOT = { any_player = { this = prev } }
				}
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE removed nonessential parentless lowborn staff attached during AI staffing window" }
				if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
				death = { death_reason = death_disappearance }
			}
			remove_character_flag = aawc_lineage_ai_staffing_window
			add_character_flag = aawc_lineage_ai_staffing_completed
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC LINEAGE closed AI staffing window; future automatic council filling disabled" }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } debug_log_scopes = yes }
		}
	}
}
