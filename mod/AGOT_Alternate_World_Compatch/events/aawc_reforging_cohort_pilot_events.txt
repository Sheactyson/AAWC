# Opt-in Stage 1A runtime harness. Only bounded player test decisions call
# these events; no on-action or production caller. The fixture rule is off by default.
# Every mutating immediate marks the whole save unsuitable for production certification.
# The first replaces the player's
# capital province; the second is deliberately read-only so a post-reload run
# cannot conceal a persistence failure by rebuilding the fixture.
aawc_reforging_stage1a.0001 = {
	type = character_event
	trigger = { has_game_rule = aawc_reforging_disposable_fixtures }
	title = aawc_reforging_stage1a.0001.t
	desc = {
		triggered_desc = {
			trigger = {
				root.capital_county.title_province = {
					aawc_reforging_cohort_test_fixture_status = 1
				}
			}
			desc = aawc_reforging_stage1a.fixture_pass
		}
		triggered_desc = {
			trigger = {
				NOT = {
					root.capital_county.title_province = {
						aawc_reforging_cohort_test_fixture_status = 1
					}
				}
			}
			desc = aawc_reforging_stage1a.fixture_fail
		}
	}
	theme = stewardship_domain_focus

	immediate = {
		if = {
			limit = { has_game_rule = aawc_reforging_disposable_fixtures }
			set_global_variable = { name = aawc_reforging_fixture_tainted value = 1 }
			root.capital_county.title_province = {
				aawc_reforging_cohort_test_initializer_effect = yes
				aawc_reforging_cohort_stage1_prepare_test_effect = yes
				aawc_reforging_cohort_read_only_diagnostics_effect = yes
			}

		}
}

	option = { name = aawc_reforging_stage1a.0001.a }
}

# Disposable post-migration regression. The called effect uses exactly two
# verified-empty provinces, exercises all nine production adapters, checks the
# paired transfer and removes all state before this result is rendered.
aawc_reforging_stage1a.0007 = {
	type = character_event
	trigger = { has_game_rule = aawc_reforging_disposable_fixtures }
	title = aawc_reforging_stage1a.0007.t
	desc = {
		triggered_desc = {
			trigger = {
				root = {
					var:aawc_reforging_production_runtime_selected = 2
					var:aawc_reforging_production_runtime_failures = 0
					var:aawc_reforging_production_runtime_before_total = var:aawc_reforging_production_runtime_after_total
				}
			}
			desc = aawc_reforging_stage1a.production_runtime_pass
		}
		triggered_desc = {
			trigger = {
				NOT = {
					root = {
						var:aawc_reforging_production_runtime_selected = 2
						var:aawc_reforging_production_runtime_failures = 0
						var:aawc_reforging_production_runtime_before_total = var:aawc_reforging_production_runtime_after_total
					}
				}
			}
			desc = aawc_reforging_stage1a.production_runtime_fail
		}
	}
	theme = stewardship_domain_focus
	immediate = {
		if = {
			limit = { has_game_rule = aawc_reforging_disposable_fixtures }
			set_global_variable = { name = aawc_reforging_fixture_tainted value = 1 }
		 aawc_reforging_cohort_stage1_production_runtime_test_effect = yes
		}
}
	option = {
		name = aawc_reforging_stage1a.0007.a
		if = {
			limit = { has_game_rule = aawc_reforging_disposable_fixtures }
			remove_variable = aawc_reforging_production_runtime_selected
			remove_variable = aawc_reforging_production_runtime_failures
			remove_variable = aawc_reforging_production_runtime_before_total
			remove_variable = aawc_reforging_production_runtime_source_after
			remove_variable = aawc_reforging_production_runtime_destination_after
			remove_variable = aawc_reforging_production_runtime_after_total
		}
	}
}
