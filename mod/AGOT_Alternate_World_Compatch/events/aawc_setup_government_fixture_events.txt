namespace = aawc_setup_government_fixture

# Manual native fixtures only. Disposable-only decisions provide UI entry.
# Use a disposable campaign with the existing default-off fixtures rule ON.
# Each immediate repeats the guard because debug-console dispatch can bypass triggers.
# Events deliberately alter real holdings/governments; the entire save is tainted.

aawc_setup_government_fixture.0001 = {
	type = character_event
	hidden = yes
	trigger = { has_game_rule = aawc_reforging_disposable_fixtures }
	immediate = {
		if = {
			limit = {
				has_game_rule = aawc_reforging_disposable_fixtures
				agot_is_dummy_character = no
				exists = capital_province
			}
			set_global_variable = { name = aawc_reforging_fixture_tainted value = 1 }
			capital_province = { set_holding_type = city_holding }
			change_government = republic_government
			if = {
				limit = { has_government = republic_government capital_province = { has_holding_type = city_holding } }
				capital_province.barony = { set_variable = { name = aawc_setup_fixture_tribal_prepared value = 2 } }
			}
			capital_province.barony = {
				aawc_setup_prepare_tribal_holding_effect = yes
				aawc_setup_apply_tribal_government_effect = yes
				# Repeat the actual boundary to exercise already-target behavior.
				aawc_setup_prepare_tribal_holding_effect = yes
				aawc_setup_apply_tribal_government_effect = yes
			}
			# Reproduce a replacement ruler selecting a secondary city capital.
			capital_county.title_province.barony = { save_scope_as = aawc_fixture_true_capital }
			capital_county = {
				random_county_province = {
					limit = {
						has_holding = yes
						barony = { is_capital_barony = no }
						NOR = { has_holding_type = ruin_holding has_holding_type = wilderness_holding has_holding_type = unknown_holding }
					}
					barony = { save_scope_as = aawc_fixture_secondary_capital }
				}
			}
			if = {
				limit = { exists = scope:aawc_fixture_secondary_capital }
				# A generated secondary may already be a city.  Avoid issuing the
				# idempotence-hostile engine effect while preserving the fixture setup.
				scope:aawc_fixture_secondary_capital.title_province = {
					if = {
						limit = { NOT = { has_holding_type = city_holding } }
						set_holding_type = city_holding
					}
				}
				create_title_and_vassal_change = { type = granted save_scope_as = aawc_fixture_capital_change add_claim_on_loss = no }
				scope:aawc_fixture_secondary_capital = {
					change_title_holder = { holder = root change = scope:aawc_fixture_capital_change }
				}
				resolve_title_and_vassal_change = scope:aawc_fixture_capital_change
				set_realm_capital = scope:aawc_fixture_secondary_capital
				change_government = republic_government
				if = {
					limit = { has_government = republic_government capital_province.barony = { is_capital_barony = no } }
					scope:aawc_fixture_true_capital = { set_variable = { name = aawc_setup_fixture_displaced_prepared value = 2 } }
				}
				aawc_setup_convert_tribal_counties_effect = yes
				aawc_setup_convert_tribal_counties_effect = yes
			}
		}
	}
}

aawc_setup_government_fixture.0003 = {
	type = character_event
	hidden = yes
	trigger = { has_game_rule = aawc_reforging_disposable_fixtures }
	immediate = {
		if = {
			limit = { has_game_rule = aawc_reforging_disposable_fixtures }
			set_global_variable = { name = aawc_reforging_fixture_tainted value = 1 }
			if = {
				limit = { exists = global_var:citadel_title }
				global_var:citadel_title = { save_scope_as = aawc_fixture_citadel }
			}
			else = { title:b_the_citadel = { save_scope_as = aawc_fixture_citadel } }
			scope:aawc_fixture_citadel.title_province = { set_holding_type = city_holding }
			create_character = {
				gender = male
				location = scope:aawc_fixture_citadel.title_province
				culture = root.culture
				faith = root.faith
				age = 45
				trait = maester
				trait = archmaester
				save_scope_as = aawc_fixture_archmaester
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = aawc_fixture_title_change
				add_claim_on_loss = no
			}
			scope:aawc_fixture_citadel = {
				change_title_holder = {
					holder = scope:aawc_fixture_archmaester
					change = scope:aawc_fixture_title_change
				}
			}
			# Exercise the actual title-gain callback, not a fixture-only government assignment.
			resolve_title_and_vassal_change = scope:aawc_fixture_title_change
		}
	}
}
