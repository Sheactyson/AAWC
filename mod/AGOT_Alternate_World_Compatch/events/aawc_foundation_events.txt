namespace = aawc_foundation

# These events deliberately use CK3's actual culture-creation effect instead of
# sending the player to the native divergence window. Faith creation remains a
# native GUI operation, but the event selects the parent religion first so the
# proper AGOT doctrine groups are available. The culture and faith invitations
# are intentionally independent so their game-rule modes do not cross-trigger.

# Each AI founder receives an individual hidden event. This makes that ruler
# root before invoking the same immediate culture finalizer proven for players.
aawc_foundation.0400 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = yes
		is_landed = yes
		agot_is_dummy_character = no
		has_character_flag = aawc_1103_ai_culture_foundation_queued
		NOT = { has_character_flag = aawc_1103_ai_culture_foundation_completed }
	}

	immediate = {
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC 1.10.3 AI individual founder culture event started" }
		set_global_variable = { name = aawc_1103_culture_namer value = root }
		aawc_1103_finalize_player_culture_foundation_effect = yes
		remove_character_flag = aawc_1103_ai_culture_foundation_queued
		add_character_flag = aawc_1103_ai_culture_foundation_completed
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC 1.10.3 AI culture foundation completed through individual founder event" }
	}
}

	theme = realm

	trigger = {
		is_landed = yes
		agot_is_dummy_character = no
		NOT = { has_character_flag = aawc_1103_culture_foundation_invited }
	}

	immediate = {
		add_character_flag = aawc_1103_culture_foundation_invited
		add_character_flag = aawc_1103_foundation_culture_pending
		set_global_variable = { name = aawc_1103_culture_namer value = root }
	}

	option = {
		name = aawc_foundation.0001.a
		# A text-widget placeholder is submitted as an empty custom value by CK3.
		# The explicit default path therefore never reads the widget at all.
		add_character_flag = aawc_1103_culture_name_uses_county
		trigger_event = aawc_foundation.0100
	}
	option = {
		name = aawc_foundation.0001.b
		trigger_event = aawc_foundation.0003
	}
}

# The custom-name path is intentionally separate from the capital-name default.
# CK3 only writes a reliable custom text value after this explicit input action.
aawc_foundation.0003 = {
	type = character_event
	title = aawc_foundation.0003.t
	desc = aawc_foundation.0003.desc
	theme = realm

	widgets = {
		widget = {
			gui = "aawc_1103_culture_name_widget"
			container = "dynamic_birth_name"
			controller = {
				type = text
				data = {
					key = aawc_1103_culture_name
					default = { localization_key = aawc_foundation_culture_default_name }
				}
			}
			setup_scope = { root = { save_scope_as = text_target } }
		}
	}

	option = {
		name = aawc_foundation.0003.a
		set_variable = { name = aawc_1103_culture_name value = flag:custom }
		trigger_event = aawc_foundation.0100
	}
	option = {
		name = aawc_foundation.0003.b
		add_character_flag = aawc_1103_culture_name_uses_county
		trigger_event = aawc_foundation.0100
	}
}

# A newly landed former adventurer may not expose a stable capital scope until
# the next daily update. Reapply the selected seed after that update.
aawc_foundation.0300 = {
	type = character_event
	hidden = yes

	trigger = {
		is_landed = yes
		has_character_flag = aawc_1103_foundation_faith_seed_selected
	}

	immediate = {
		every_courtier = { set_character_faith = root.faith }
		capital_county = { set_county_faith = root.faith }
		if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC 1.10.4.6 delayed player faith seed synchronization completed" }
	}
}

	}
}
