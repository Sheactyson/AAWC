# One-day delayed startup reconciliation runs after the native/AGOT world-mode
# conversion children have removed holdings or assigned wilderness types.
aawc_population_event.0101 = {
	type = character_event
	hidden = yes
	immediate = {
		aawc_reforging_cohort_reconcile_world_settlement_eligibility_effect = yes
		aawc_population_recover_missing_monthly_schedulers_effect = yes
	}
}

aawc_population_event.0100 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				agot_is_dummy_character = no
				is_landed = yes
				NOT = { has_character_flag = aawc_population_monthly_tick_recent }
			}
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC POP 1.9 monthly event tick applying growth" }
			aawc_no_guest_clear_host_effect = yes
			add_character_flag = { flag = aawc_population_monthly_tick_recent days = 27 }
			aawc_population_apply_monthly_growth_character_effect = yes
			trigger_event = { id = aawc_population_event.0100 days = 31 }
			set_variable = { name = aawc_population_monthly_tick_scheduled value = 1 }
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC POP 1.9 monthly event tick rescheduled" }
		}
		else_if = {
			limit = {
				agot_is_dummy_character = no
				is_landed = yes
				has_character_flag = aawc_population_monthly_tick_recent
			}
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC POP 1.9 duplicate monthly event skipped without reschedule" }
		}
		else = {
			if = { limit = { aawc_diagnostics_verbose_enabled = yes } error_log = "AAWC POP 1.9 monthly event reached invalid/non-landed scope and was not rescheduled" }
			remove_variable = aawc_population_monthly_tick_scheduled
		}
	}
}
